class Node:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None


def inorder(root):
    if root:
        inorder(root.left)
        print(root.data, end=" -> ")
        inorder(root.right)


def preorder(root):
    if root:
        print(root.data, end=" -> ")
        preorder(root.left)
        preorder(root.right)


def postorder(root):
    if root:
        postorder(root.left)
        postorder(root.right)
        print(root.data, end=" -> ")


# Library Catalog Tree

root = Node("Library")

root.left = Node("Science")
root.right = Node("Technology")

root.left.left = Node("Physics")
root.left.right = Node("Chemistry")

root.right.left = Node("Programming")
root.right.right = Node("Electronics")


print("Preorder Catalog:")
preorder(root)

print("\n\nInorder Catalog:")
inorder(root)

print("\n\nPostorder Catalog:")
postorder(root)
