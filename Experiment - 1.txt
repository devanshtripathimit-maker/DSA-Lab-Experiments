stack = []

def push(book):
    stack.append(book)
    print(book, "returned and added to stack.")

def pop():
    if not stack:
        print("No books to issue.")
    else:
        print("Book issued:", stack.pop())

def display():
    if not stack:
        print("Stack is empty.")
    else:
        print("Returned books:")
        for book in reversed(stack):
            print(book)

while True:
    print("\n1. Return Book")
    print("2. Issue Book")
    print("3. Display Books")
    print("4. Exit")

    choice = int(input("Enter choice: "))

    if choice == 1:
        book = input("Enter book name: ")
        push(book)

    elif choice == 2:
        pop()

    elif choice == 3:
        display()

    elif choice == 4:
        break

    else:
        print("Invalid choice")
