queue = []
front = 0

def enqueue(name):
    queue.append(name)
    print(name, "joined the queue.")

def dequeue():
    global front

    if front == len(queue):
        print("Queue is empty.")
    else:
        print("Ticket booked for:", queue[front])
        front += 1

def display():
    if front == len(queue):
        print("Queue is empty.")
    else:
        print("People in queue:")
        for i in range(front, len(queue)):
            print(queue[i])

while True:
    print("\n1. Join Queue")
    print("2. Book Ticket")
    print("3. Display Queue")
    print("4. Exit")

    choice = int(input("Enter choice: "))

    if choice == 1:
        name = input("Enter name: ")
        enqueue(name)

    elif choice == 2:
        dequeue()

    elif choice == 3:
        display()

    elif choice == 4:
        break

    else:
        print("Invalid choice")
